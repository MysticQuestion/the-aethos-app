### 2. Add `services/calculator/app/timezone.py`
 `from __future__ import annotations  from dataclasses import dataclass from datetime import datetime, timezone from typing import Literal, Optional  from dateutil import tz  TzInvalidPolicy = Literal["error", "normalize_forward"] TzAmbiguousPolicy = Literal["prefer_earliest", "prefer_latest"] TzResolution = Literal["ok", "ambiguous", "invalid", "time_unknown"]   @dataclass(frozen=True) class TzResolutionResult:     dt_local: datetime     dt_utc: datetime     resolution: TzResolution     note: Optional[str] = None   def resolve_local_to_utc(     date_yyyy_mm_dd: str,     time_hh_mm_ss: str,     tz_name: str,     invalid_policy: TzInvalidPolicy = "normalize_forward",     ambiguous_policy: TzAmbiguousPolicy = "prefer_earliest", ) -> TzResolutionResult:     """     Converts local date+time in IANA tz to UTC with explicit DST policies.     - Handles nonexistent times (spring-forward) via invalid_policy     - Handles ambiguous times (fall-back) via ambiguous_policy using fold     """     zone = tz.gettz(tz_name)     if zone is None:         raise ValueError(f"Unknown IANA timezone: {tz_name}")      dt_naive = datetime.fromisoformat(f"{date_yyyy_mm_dd}T{time_hh_mm_ss}")     dt_local = dt_naive.replace(tzinfo=zone)      # Nonexistent time (spring forward)     if not tz.datetime_exists(dt_local):         if invalid_policy == "error":             raise ValueError(f"Invalid local time (nonexistent due to DST): {dt_local.isoformat()}")         # normalize forward to the next real time         dt_local = tz.resolve_imaginary(dt_local)         return TzResolutionResult(             dt_local=dt_local,             dt_utc=dt_local.astimezone(timezone.utc),             resolution="invalid",             note="nonexistent_local_time_normalized_forward"         )      # Ambiguous time (fall back) — occurs twice; choose fold     if tz.datetime_ambiguous(dt_local):         # fold=0 => first occurrence; fold=1 => second occurrence         fold = 0 if ambiguous_policy == "prefer_earliest" else 1         dt_local = dt_local.replace(fold=fold)         return TzResolutionResult(             dt_local=dt_local,             dt_utc=dt_local.astimezone(timezone.utc),             resolution="ambiguous",             note=f"ambiguous_local_time_fold={fold}"         )      return TzResolutionResult(         dt_local=dt_local,         dt_utc=dt_local.astimezone(timezone.utc),         resolution="ok"     ) `
