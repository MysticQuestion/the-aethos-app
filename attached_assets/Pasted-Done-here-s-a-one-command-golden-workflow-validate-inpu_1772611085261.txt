from __future__ import annotations

import hashlib
import json
from datetime import datetime, timezone
from typing import Any, Dict, List, Tuple

SIGNS = [
    "Aries","Taurus","Gemini","Cancer","Leo","Virgo",
    "Libra","Scorpio","Sagittarius","Capricorn","Aquarius","Pisces"
]

BODIES = [
    "Sun","Moon","Mercury","Venus","Mars","Jupiter","Saturn",
    "Uranus","Neptune","Pluto","NorthNode","SouthNode","Chiron"
]

ASPECTS = [
    ("conjunction", 0.0),
    ("sextile", 60.0),
    ("square", 90.0),
    ("trine", 120.0),
    ("opposition", 180.0),
    ("quincunx", 150.0)
]

def stable_hash(obj: Any) -> str:
    s = json.dumps(obj, sort_keys=True, separators=(",", ":"))
    return hashlib.sha256(s.encode("utf-8")).hexdigest()

def utc_now_rfc3339() -> str:
    return datetime.now(timezone.utc).replace(microsecond=0).isoformat().replace("+00:00", "Z")

def clamp360(x: float) -> float:
    x = x % 360.0
    return x if x >= 0 else x + 360.0

def sign_from_lon(lon: float) -> str:
    idx = int(clamp360(lon) // 30.0)
    return SIGNS[idx]

def house_from_lon_whole_sign(asc_lon: float, lon: float) -> int:
    asc_sign = int(clamp360(asc_lon) // 30.0)
    body_sign = int(clamp360(lon) // 30.0)
    # Whole sign houses: Asc sign = House 1
    delta = (body_sign - asc_sign) % 12
    return delta + 1

def placeholder_angles(seed: int) -> Tuple[float, float]:
    # Deterministic "Asc" and "MC" from seed
    asc = clamp360((seed % 360000) / 1000.0)
    mc = clamp360(asc + 90.0)
    return asc, mc

def placeholder_body_lon(seed: int, idx: int) -> float:
    # Spread bodies around the wheel deterministically
    base = (seed % 360000) / 1000.0
    return clamp360(base + idx * 23.173)

def placeholder_speed(idx: int) -> float:
    # Deterministic pseudo-speed
    return 0.1 + (idx % 7) * 0.03

def build_houses(asc_lon: float, house_system: str) -> List[Dict[str, Any]]:
    houses = []
    # For stub: implement whole-sign faithfully; quadrant systems get placeholders
    if house_system == "whole_sign":
        asc_sign_idx = int(clamp360(asc_lon) // 30.0)
        for h in range(1, 13):
            sign_idx = (asc_sign_idx + (h - 1)) % 12
            start = sign_idx * 30.0
            end = clamp360(start + 30.0)
            houses.append({
                "house": h,
                "sign": SIGNS[sign_idx],
                "start_lon": start,
                "end_lon": end,
                "cusp_lon": start
            })
    else:
        # Placeholder cusps: evenly spaced
        for h in range(1, 13):
            cusp = clamp360(asc_lon + (h - 1) * 30.0)
            start = cusp
            end = clamp360(cusp + 30.0)
            houses.append({
                "house": h,
                "sign": sign_from_lon(cusp),
                "start_lon": start,
                "end_lon": end,
                "cusp_lon": cusp
            })
    return houses

def compute_aspects(bodies: List[Dict[str, Any]], orb_policy: Dict[str, Any]) -> List[Dict[str, Any]]:
    # Stub aspect detection: compute separations and include major aspects within configured orbs
    # Deterministic: order by body indices.
    orb_map = orb_policy["major_aspects"]
    def orb_for(aspect: str) -> float:
        if aspect in orb_map:
            return float(orb_map[aspect])
        # quincunx default tight
        return 3.0

    out = []
    for i in range(len(bodies)):
        for j in range(i+1, len(bodies)):
            a = bodies[i]
            b = bodies[j]
            lon_a = float(a["lon"])
            lon_b = float(b["lon"])
            sep = abs(lon_a - lon_b) % 360.0
            sep = min(sep, 360.0 - sep)
            for name, exact in ASPECTS:
                orb = abs(sep - exact)
                if orb <= orb_for(name):
                    strength = max(0.0, 1.0 - (orb / max(orb_for(name), 0.0001)))
                    out.append({
                        "body_a": a["body"],
                        "body_b": b["body"],
                        "aspect": name,
                        "exact_angle": exact,
                        "separation": sep,
                        "orb": orb,
                        "applying": False,  # stub
                        "strength": round(strength, 6)
                    })
    return out

def derived_sect(sun_lon: float, asc_lon: float) -> str:

    # ---- Handle unknown time
    if req.birth.birth_local_time is None:

        # but mark reliability and tz_resolution accordingly.

        tz_resolution = "time_unknown"

        time_uncertainty = req.birth.time_accuracy_minutes or 1440

        birth_local_time = req.birth.birth_local_time

        reliability_flags = []
        time_uncertainty = req.birth.time_accuracy_minutes or 0

        "birth_local_date": req.birth.birth_local_date,
        "birth_local_time": birth_local_time,
        "birth_timezone": req.birth.birth_timezone,

    generated_at = utc_now_rfc3339()

    birth_utc = f"{req.birth.birth_local_date}T{birth_local_time}Z"

            house = house_from_lon_whole_sign(asc_lon, lon)

            "lon": float(lon),

            "distance_au": 1.0,
            "speed_lon": sp,
            "speed_lat": 0.0,
            "speed_dist": 0.0,

            "sign": sign,

        })

        "sect": sect,
        "dignities": dignities_stub(bodies),
        "angularity": angularity_stub(bodies)

            }

            "birth_local_date": req.birth.birth_local_date,
            "birth_local_time": birth_local_time,
            "birth_timezone": req.birth.birth_timezone,
            "birth_utc_datetime": birth_utc,
            "julian_day": float(julian_day),
            "delta_t_seconds": float(delta_t),
            "geo": {
                "lat": float(req.birth.geo.lat),
                "lon": float(req.birth.geo.lon),
                "place_label": req.birth.geo.place_label
            },
            "tz_invalid_time_policy": tz_invalid_time_policy,
            "tz_ambiguous_time_policy": tz_ambiguous_time_policy,
            "tz_resolution": tz_resolution
        },
        "config": req.config.model_dump(),
        "western_chart": {
            "zodiac": req.config.zodiac_system,
            "ayanamsa": req.config.ayanamsa,
            "house_system": req.config.house_system,
            "angles": {
                "asc_lon": float(asc_lon),
                "mc_lon": float(mc_lon),
                "desc_lon": float((asc_lon + 180.0) % 360.0),
                "ic_lon": float((mc_lon + 180.0) % 360.0)
            },
            "bodies": bodies,
            "houses": houses

        "aspects": aspects,
        "derived": derived,

            "time_uncertainty_minutes": int(time_uncertainty),
            "flags": reliability_flags if reliability_flags else None

    }

    # Pydantic response_model validation (ensures shape stays honest)
    try:
        return CanonicalOutput.model_validate(out)
